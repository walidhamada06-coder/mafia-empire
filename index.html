-- MAFIA EMPIRE - Phase 1 -- Run this once in Supabase SQL Editor. -- This creates a separate operations table and a secure server-side completion function. create table if not exists public.player_operations ( id uuid primary key default gen_random_uuid(), player_id uuid not null references public.players(id) on delete cascade, operation_type text not null check (operation_type in ('bike','car')), started_at timestamptz not null default now(), finish_at timestamptz not null, status text not null default 'running' check (status in ('running','completed','failed')), reward integer not null default 0, xp_reward integer not null default 0, wanted_gain integer not null default 0, created_at timestamptz not null default now() ); create index if not exists player_operations_player_idx on public.player_operations(player_id, status); alter table public.player_operations enable row level security; drop policy if exists "players can read own operations" on public.player_operations; create policy "players can read own operations" on public.player_operations for select to authenticated using (player_id = auth.uid()); drop policy if exists "players can create own operations" on public.player_operations; create policy "players can create own operations" on public.player_operations for insert to authenticated with check (player_id = auth.uid()); create or replace function public.finish_player_operation(p_operation_id uuid) returns jsonb language plpgsql security definer set search_path = public as $$ declare op public.player_operations; p public.players; begin select * into op from public.player_operations where id = p_operation_id and player_id = auth.uid() for update; if not found then raise exception 'Operation not found'; end if; if op.status <> 'running' then return jsonb_build_object('ok', false, 'message', 'Operation already finished'); end if; if now() < op.finish_at then raise exception 'Operation is not finished yet'; end if; select * into p from public.players where id = auth.uid() for update; update public.players set money = coalesce(money,0) + op.reward, xp = coalesce(xp,0) + op.xp_reward, wanted_level = coalesce(wanted_level,0) + op.wanted_gain where id = auth.uid(); update public.player_operations set status = 'completed' where id = op.id; return jsonb_build_object( 'ok', true, 'reward', op.reward, 'xp', op.xp_reward, 'wanted_gain', op.wanted_gain ); end; $$; revoke all on function public.finish_player_operation(uuid) from public; grant execute on function public.finish_player_operation(uuid) to authenticated;
